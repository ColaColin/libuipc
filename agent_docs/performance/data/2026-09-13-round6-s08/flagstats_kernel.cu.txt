    // round-6 (s08): DIAGNOSIS ONLY (`UIPC_CONTACT_FLAGSTATS=1`). A census of
    // the distance-flag dimension of every PT / EE / PE / PP pair the contact
    // assembler sees, plus the EE mollifier split. It decides where part 1's
    // projection cost actually sits: `*_barrier_make_spd` dispatches on
    // `active_count(flag)` (2 = the pair degenerates to point-point, 3 = to
    // point-edge, 4 = the interior case), and those three branches run
    // completely different code. Never launched unless the env var is set.
    __global__ void contact_flag_stats_kernel(cuda_tool::CBufferView<Vector3>  Ps,
                                              cuda_tool::CBufferView<Vector3> rest_Ps,
                                              cuda_tool::CBufferView<Float> thicknesses,
                                              cuda_tool::CBufferView<Float> d_hats,
                                              cuda_tool::BufferView<unsigned long long> dmin,
                                              cuda_tool::CBufferView<Vector4i> PTs,
                                              cuda_tool::CBufferView<Vector4i> EEs,
                                              cuda_tool::CBufferView<Vector3i> PEs,
                                              cuda_tool::CBufferView<Vector2i> PPs,
                                              cuda_tool::BufferView<unsigned long long> ctr,
                                              IndexT ee_offset,
                                              IndexT pe_offset,
                                              IndexT pp_offset,
                                              int    n)
    {
        int idx = blockIdx.x * blockDim.x + threadIdx.x;
        if(idx >= n)
            return;
        using namespace distance;
        using namespace sym::codim_ipc_simplex_contact;
        if(idx < ee_offset)  // PT
        {
            Vector4i PT   = PTs(idx);
            Vector4i flag = point_triangle_distance_flag(
                Ps(PT[0]), Ps(PT[1]), Ps(PT[2]), Ps(PT[3]));
            atomicAdd(&ctr(detail::active_count(flag) - 2), 1ull);
            if(detail::active_count(flag) == 4)
            {
                Float D;
                point_triangle_distance2(flag, Ps(PT[0]), Ps(PT[1]), Ps(PT[2]), Ps(PT[3]), D);
                Float th = PT_thickness(thicknesses(PT(0)), thicknesses(PT(1)),
                                        thicknesses(PT(2)), thicknesses(PT(3)));
                Float dh = PT_d_hat(d_hats(PT(0)), d_hats(PT(1)), d_hats(PT(2)), d_hats(PT(3)));
                Float r  = D / ((dh + th) * (dh + th));
                atomicMin(&dmin(0), __double_as_longlong(r));
            }
            return;
        }
        if(idx < pe_offset)  // EE
        {
            int      i  = idx - ee_offset;
            Vector4i EE = EEs(i);
            Vector4i flag =
                edge_edge_distance_flag(Ps(EE[0]), Ps(EE[1]), Ps(EE[2]), Ps(EE[3]));
            Float eps_x;
            edge_edge_mollifier_threshold(rest_Ps(EE[0]),
                                          rest_Ps(EE[1]),
                                          rest_Ps(EE[2]),
                                          rest_Ps(EE[3]),
                                          static_cast<Float>(1e-3),
                                          eps_x);
            Float cross_norm2;
            edge_edge_cross_norm2(Ps(EE[0]), Ps(EE[1]), Ps(EE[2]), Ps(EE[3]), cross_norm2);
            bool mollified = cross_norm2 < eps_x;
            atomicAdd(&ctr(3 + (mollified ? 3 : 0) + detail::active_count(flag) - 2), 1ull);
            if(!mollified && detail::active_count(flag) == 4)
            {
                Float D;
                edge_edge_distance2(flag, Ps(EE[0]), Ps(EE[1]), Ps(EE[2]), Ps(EE[3]), D);
                Float th = EE_thickness(thicknesses(EE(0)), thicknesses(EE(1)),
                                        thicknesses(EE(2)), thicknesses(EE(3)));
                Float dh = EE_d_hat(d_hats(EE(0)), d_hats(EE(1)), d_hats(EE(2)), d_hats(EE(3)));
                Float r  = D / ((dh + th) * (dh + th));
                atomicMin(&dmin(1), __double_as_longlong(r));
            }
            return;
        }
        if(idx < pp_offset)  // PE
        {
            int      i    = idx - pe_offset;
            Vector3i PE   = PEs(i);
            Vector3i flag = point_edge_distance_flag(Ps(PE[0]), Ps(PE[1]), Ps(PE[2]));
            atomicAdd(&ctr(9 + detail::active_count(flag) - 2), 1ull);
            return;
        }
        atomicAdd(&ctr(11), 1ull);  // PP
    }
